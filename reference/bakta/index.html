    def assign_annotation_confidence(entry: dict, prostt5_confidence, structures: bool) -> str:
    """Classify a Foldseek hit into a confidence tier: 'high', 'medium' or 'low'.

    Mirrors phold's heuristic (phold.subcommands.compare.assign_annotation_confidence).
    ``entry`` is the hierarchy-selected pstc hit carrying ``query_cov`` (qCov),
    ``subject_cov`` (tCov), ``identity`` (fident) and ``evalue``.
    ``prostt5_confidence`` is the protein's mean ProstT5 confidence (0-100), or
    None for structure input. With ``structures`` (or no ProstT5 confidence
    available) the ProstT5 criteria are dropped.
    """
    qcov = entry.get('query_cov', 0.0)
    tcov = entry.get('subject_cov', 0.0)
    fident = entry.get('identity', 0.0)
    try:
        evalue = float(entry.get('evalue', 1.0))
    except (TypeError, ValueError):
        evalue = 1.0

    # Thresholds as documented in docs/output.md ("Annotation confidence"):
    # coverage >= 0.8, identity > 0.3, ProstT5 >= 60 (high) / >= 45 (medium),
    # E-value < 1e-10 (high) / < 1e-5 (medium, always required).
    cov_both = qcov >= 0.8 and tcov >= 0.8
    cov_any = qcov >= 0.8 or tcov >= 0.8
    light_zone = fident > 0.3

    if structures or prostt5_confidence is None:
        if cov_both and (light_zone or evalue < 1e-10):
            return 'high'
        # Medium needs E < 1e-5 on both paths (previously OR'd here, so a
        # structure hit at E = 1e-4 could be 'medium' contrary to the docs).
        if cov_any and light_zone and evalue < 1e-5:
            return 'medium'
        return 'low'

    # ProstT5 path: mean 3Di confidence factors in.
    if cov_both and (light_zone or prostt5_confidence >= 60 or evalue < 1e-10):
        return 'high'
    # '>= 45', not '45 <= x <= 60': confidence above 60 must not drop a hit
    # from medium to low (the old band made the tier non-monotonic).
    if cov_any and (light_zone or prostt5_confidence >= 45) and evalue < 1e-5:
        return 'medium'
    return 'low'

    def attach_prostt5_confidence(features: Sequence[dict], mean_probs_csv, has_duplicate_locus: bool = False) -> None:
    """Attach each CDS's mean ProstT5 confidence (0-100) from the
    ``{prefix}_prostT5_3di_mean_probabilities.csv`` file (``cds_id,mean_prob``,
    no header) as ``feature['prostt5_confidence']``. No-op if the file is absent
    (e.g. structure input)."""
    confidence = {}
    try:
        with open(mean_probs_csv) as fh:
            for line in fh:
                parts = line.rstrip('\n').split(',')
                if len(parts) >= 2:
                    try:
                        confidence[parts[0]] = float(parts[1])
                    except ValueError:
                        continue
    except FileNotFoundError:
        return

    for feat in features:
        key = feat.get('id') if has_duplicate_locus else feat.get('locus')
        if key in confidence:
            feat['prostt5_confidence'] = confidence[key]

    def cath_domain_label(name: str) -> str:
    """Turn a CATH superfamily name into a domain-level product name.

    A CATH hit is evidence that the protein contains a domain of that
    superfamily, not that the whole protein has the named function, so the
    label says so explicitly (the NCBI/Bakta convention for domain-only
    evidence): "SH2 domain" -> "SH2 domain-containing protein",
    "Enoyl acyl carrier protein reductase" -> "Enoyl acyl carrier protein
    reductase domain-containing protein". Representative-specific suffixes
    ("; Chain A", ", domain 2", "from Thermus thermophilus HB27") are removed.
    """
    label = name.strip().rstrip('.').strip()
    label = RE_CATH_CHAIN_SUFFIX.sub('', label)
    label = RE_CATH_DOMAIN_NUMBER.sub('', label)
    label = RE_CATH_FROM_SPECIES.sub('', label)
    label = label.strip().rstrip(',;').strip()
    if not label:
        return label
    if re.search(r'\bdomains?$', label, flags=re.IGNORECASE):
        label = re.sub(r'\bdomains$', 'domain', label, flags=re.IGNORECASE)
        return f"{label}-containing protein"
    return f"{label} domain-containing protein"

    def extract_protein_gene_symbol(product: str) -> str:
    """
    Extracts a valid gene symbol from a protein name.

    Args:
      product (str): The protein name to extract a gene symbol from.

    Returns:
      str: The extracted gene symbol.

    Examples:
      >>> extract_protein_gene_symbol(product)
    """
    gene_symbols = []
    for part in product.split(' '):  # try to extract valid gene symbols
        m = RE_GENE_SYMBOL.fullmatch(part)
        if(m):
            symbol = m[0]
            logger.info('fix gene: extract symbol from protein name. symbol=%s', symbol)
            gene_symbols.append(symbol)
        else:
            m = RE_PROTEIN_SYMBOL.fullmatch(part)  # extract protein names
            if(m):
                symbol = m[0]
                symbol = symbol[0].lower() + symbol[1:]
                logger.info('fix gene: extract symbol from protein name. symbol=%s', symbol)
                gene_symbols.append(symbol)
    if(len(gene_symbols) == 0):  # None found
        return None
    elif(len(gene_symbols) == 1):  # found 1
        return gene_symbols[0]
    else:  # found more than one, take the 2nd as the 1st often describes a broader gene family like "xyz family trancsriptional regulator ..."
        return gene_symbols[1]

    def is_informative_product(product) -> bool:
    """True if ``product`` names a function, i.e. is not empty and not a
    hypothetical/uncharacterised placeholder (before or after Bakta's product
    revision)."""
    if not product:
        return False
    if RE_UNINFORMATIVE_PRODUCT.search(product):
        return False
    # revise_cds_product can also blank out suspect products (contig names,
    # no letters, pure tmRNA), which must fall through too. Run it quietly:
    # the selected product is revised (and logged) once more by the caller.
    logger.disable(__name__)
    try:
        revised = revise_cds_product(product)
    finally:
        logger.enable(__name__)
    return bool(revised) and not RE_UNINFORMATIVE_PRODUCT.search(revised)

    def pstc_db_xrefs(entry, fast: bool, selected_entry=None) -> list:
    """The db_xrefs that combine_annotation adds for one PSTC hit.

        return [f"cath:cath_{eid}"]

    def reset_baktfold_annotation(feature: dict, keep_product: bool = False) -> bool:
    """Undo combine_annotation on a CDS so that label selection can be rerun
    on its stored hits (``baktfold json --relabel``).

    Only features Baktfold labelled (``baktfold`` key, set whenever a CDS had
    hits) are reset: the product goes back to "hypothetical protein", the
    db_xrefs that came from its hits are removed, and the confidence tier and
    structure scores are dropped. The hits themselves (``pstc``) are kept.
    Must run before the hits are changed (chains or descriptions refreshed),
    because the xrefs to remove are rebuilt from them.

    possible with ``--all-proteins``) cannot be recovered. With
    ``keep_product`` (a curated input label that label selection kept: hits,
    but no confidence tier), the product is kept as the starting state, as
    ``run`` starts from it.

    Returns True if the feature was reset.
    """
    if not feature.get('baktfold'):
        return False
    pstc = feature.get('pstc') or []
    pstc = pstc if isinstance(pstc, list) else [pstc]
    added = set()
    for entry in pstc:
        # fast=False and selected=entry: every xref any hit could have added
        added.update(pstc_db_xrefs(entry, False, entry))
    feature['db_xrefs'] = [x for x in feature.get('db_xrefs', []) if x not in added]
    # back to the state combine_annotation started from. Not mark_as_hypothetical:
    # that also clears gene/genes, which combine_annotation leaves alone
    if not keep_product:
        feature['product'] = bc.HYPOTHETICAL_PROTEIN
        feature['hypothetical'] = True
    for key in ('annotation_confidence', 'tmscore', 'lddt', 'baktfold'):
        feature.pop(key, None)
    return True

    def parse(features: Sequence[dict], foldseek_df: pl.DataFrame, db_name: str = 'swissprot', has_duplicate_locus: bool = False) -> None:
    """Update CDS in place with PSTC hits from foldseek_df if they pass filters.

    has_duplicate_locus - some euks have multiple CDS per locus tag

    """

    if foldseek_df.is_empty():
        return features

    # each query maps to a list of hit rows (to handle multiple CATH greedy
    # tophits for multidomain proteins). Single pass over the rows as dicts.
    foldseek_hits = defaultdict(list)
    for row in foldseek_df.iter_rows(named=True):
        foldseek_hits[row['query']].append(row)

    updated_count = 0


    for cds in features:
        if has_duplicate_locus:
            aa_identifier = cds.get('id')
        else:
            aa_identifier = cds.get('locus')

        if aa_identifier not in foldseek_hits:
            continue  # no hits, skip

        cds_updated = False  

        # Iterate over *all* hits for this query
        for row in foldseek_hits[aa_identifier]:
            query_cov = float(row['qCov'])
            subject_cov = float(row['tCov'])
            identity = float(row['fident'])
            evalue = float(row['evalue'])
            bitscore = float(row['bitscore'])
            target_id = row['target']

            # Extract accession depending on database
            if db_name in {"swissprot", "afdb"}:
                accession = target_id.split('-')[1]
            elif db_name == "pdb":
                accession = target_id.split('-')[0]
            else:  # cath and custom
                accession = target_id

            # Apply your filters
            if (
                query_cov >= bc.MIN_PSTC_QCOVERAGE
                and subject_cov >= bc.MIN_PSTC_TCOVERAGE
                and identity >= bc.MIN_PSTC_IDENTITY
            ):
                new_pstc = {
                    'source': db_name,
                    'id': accession,
                    'query_cov': query_cov,
                    'subject_cov': subject_cov,
                    'identity': identity,
                    'score': bitscore,
                    'evalue': evalue,
                }
                # structure-based Foldseek also reports TM-score and LDDT
                # (get_tophit adds these columns only when structures=True)
                if 'alntmscore' in row and row['alntmscore'] is not None:
                    new_pstc['tmscore'] = float(row['alntmscore'])
                if 'lddt' in row and row['lddt'] is not None:
                    new_pstc['lddt'] = float(row['lddt'])
                # Foldseek homology probability, present only with --report-prob
                if 'prob' in row and row['prob'] is not None:
                    new_pstc['prob'] = float(row['prob'])
                # PDB hits are chains; keep which one, so the label can name
                # that chain's protein rather than the whole entry's title.
                if db_name == "pdb":
                    chain = pdb_chain_from_target(target_id)
                    if chain:
                        new_pstc['chain'] = chain

                # Append or initialize 'pstc'
                if 'pstc' in cds:
                    if isinstance(cds['pstc'], dict):
                        cds['pstc'] = [cds['pstc'], new_pstc]
                    elif isinstance(cds['pstc'], list):
                        cds['pstc'].append(new_pstc)
                    else:
                        cds['pstc'] = [new_pstc]
                else:
                    cds['pstc'] = [new_pstc]  # ← ensure list, since we may have many hits


                cds_updated = True  

        # Increment only once per CDS that had at least one valid hit (CATH might have multiple)
        if cds_updated:
            updated_count += 1

    logger.info(f"PSTC for {db_name} updated in place for {updated_count} CDSs")
    return features

    """Chain ID of a Foldseek PDB target, e.g. '6xgw-assembly1.cif.gz_A' -> 'A'.

    Symmetry copies in biological assemblies carry a copy suffix

    if '_' not in target_id:

    return chain or None

    def read_pdb_tophit_chains(tophit_tsv: Path) -> dict:
    """Query -> PDB chain from a ``<prefix>_pdb_tophit.tsv``.

    chains = {}

    def refresh_descriptions(features: Sequence[dict], baktfold_db: Path, pdb_chains: dict = None,
                         has_duplicate_locus: bool = False) -> dict:
    """Re-resolve the stored hits' descriptions from ``baktfold_db``, as
    ``baktfold json --relabel -d`` does before label selection is rerun.

    PDB hits without a chain take one from ``pdb_chains`` (query -> chain,
    see read_pdb_tophit_chains), which is also written back to the hit.

    Two kinds of description are kept as stored rather than looked up:
      - custom_db hits: their descriptions came from the user's annotations
        file, not from baktfold.db;
      - PDB hits with no recoverable chain, when the database has chain-level
        names: these keep the entry title they were labelled with. A lookup
        would find no chain and drop the hit, which is right for an unlisted
        chain but not for a chain that simply wasn't recorded.

    Returns counts: pdb_chains_added, pdb_no_chain_kept.
    """
    pdb_chains = pdb_chains or {}
    counts = {'pdb_chains_added': 0, 'pdb_no_chain_kept': 0}

    conn = sqlite3.connect(f"file:{Path(baktfold_db).joinpath('baktfold.db')}?mode=ro", uri=True)
    try:
        chain_level_pdb = has_table(conn, "pdb_chain")
    finally:
        conn.close()

    kept = []  # (entry, description) restored after lookup_sql
    for feat in features:
        pstc = feat.get("pstc")
        if not pstc:
            continue
        key = feat.get('id') if has_duplicate_locus else feat.get('locus')
        for entry in (pstc if isinstance(pstc, list) else [pstc]):
            if not isinstance(entry, dict):
                continue
            source = entry.get("source")
            if source == "custom_db":
                kept.append((entry, entry.get("description")))
            elif source == "pdb" and not entry.get("chain"):
                chain = pdb_chains.get(key)
                if chain:
                    entry["chain"] = chain
                    counts['pdb_chains_added'] += 1
                elif chain_level_pdb:
                    kept.append((entry, entry.get("description")))
                    counts['pdb_no_chain_kept'] += 1

    lookup_sql(features, Path(baktfold_db), threads=1)
    for entry, description in kept:
        if description:
            entry["description"] = description
    return counts
